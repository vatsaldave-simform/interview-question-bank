FROM questions q
JOIN question_tags qt ON qt.question_id = q.id
JOIN tags t ON t.id = qt.tag_id
WHERE (q.client_id IS NULL
       OR EXISTS (SELECT 1 FROM permission_grants g
                  WHERE g.viewer_id = 1 AND g.client_id = q.client_id))
  AND qt.tag_id = ANY (ARRAY[1,2,41,42,156,158])
GROUP BY q.id
HAVING count(DISTINCT t.category_id) = 3
ORDER BY q.created_at DESC, q.id DESC LIMIT 50 OFFSET 0
